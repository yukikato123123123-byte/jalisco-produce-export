import {useRef,useState} from 'react';
import {accountRequest} from './AccountPages';
import {AdminPagination,type PageInfo} from './AdminPagination';
import {operationActions} from './NopalOperationsPanels';
import {ProductImages} from './ProductImages';
import {parseProductImages} from '../productImages';
import {useLanguage,Localized} from '../localization';
type Activity=Record<string,string|number|null>;
export type ApprovalSummary={kind:string;pending:number;approved:number;rejected:number};
type Props={operations?:{counts?:Record<string,number>;recent?:Activity[];recentPagination?:PageInfo};activityTrend?:{day:string;count:number}[]|null;counts:Record<string,number>;health:Record<string,string>;recent:Activity[];recentProducts?:Activity[];approvals?:ApprovalSummary[];inquiries?:Record<string,number>;observed_at?:string;select:(section:string,status?:string)=>void};
const kinds=[['producer','Producers'],['product','Products'],['buyer','Buyers']];
const states=['pending','approved','rejected'] as const;
const valid=(v:unknown):v is number=>typeof v==='number'&&Number.isFinite(v)&&v>=0;
const statusNames:Record<string,string>={pending:'Pending',approved:'Approved',rejected:'Rejected',received:'Received',reviewing:'Under review',resolved:'Resolved',public:'Public',private:'Private',draft:'Draft'};
const actionNames:Record<string,string>={producer_review:'Producer review',product_review:'Product review',buyer_review:'Buyer review',inquiry_status:'Inquiry status change',inquiry_reply:'Inquiry reply',product_status:'Product publication',buyer_unpublish:'Buyer publication withdrawn'};
export function AdminDashboard({operations,counts,health,recent,recentProducts,approvals,inquiries,observed_at,activityTrend,select}:Props){
 const {language}=useLanguage(),locale=language==='es'?'es-MX':'en-US';
 const number=(v:unknown)=>valid(v)?v.toLocaleString(locale):'Not queried';
 const time=(v:unknown)=>typeof v==='string'&&Number.isFinite(Date.parse(v))?new Intl.DateTimeFormat(locale,{dateStyle:'medium',timeStyle:'short',timeZone:'UTC'}).format(new Date(v))+' UTC':'Not queried';
 const approvalData=kinds.map(([kind,label])=>({kind,label,data:approvals?.find(r=>r.kind===kind)}));
 const approvalsReady=approvalData.every(({data})=>data&&states.every(s=>valid(data[s])));
 const approvalMax=Math.max(1,...approvalData.map(({data})=>data?states.reduce((a,s)=>a+data[s],0):0));
 const inquiryReady=['received','reviewing','resolved'].every(s=>valid(inquiries?.[s]));
 const inquiryMax=Math.max(1,...Object.values(inquiries??{}));
 const metrics=[['products','Products','products'],['launch_list','Launch List','launch-list'],['customer_feedback','Customer Feedback','customer-feedback'],['new_inquiries','New Inquiries','inquiries'],['pending_reviews','Pending Reviews','priority']];
 const [activity,setActivity]=useState<{items:Activity[]}&PageInfo>({items:operations?.recent??recent,...operations?.recentPagination}),[products,setProducts]=useState<({items:Activity[]}&PageInfo)|null>(null),[listBusy,setListBusy]=useState(false),[listError,setListError]=useState('');
 const listLocks=useRef(new Set<string>());
 async function loadList(kind:'activity-log'|'products',page=1){if(listLocks.current.has(kind))return;listLocks.current.add(kind);setListBusy(true);setListError('');try{const value=await accountRequest(`/api/admin/${kind}?sort=updated&page=${page}`);if(kind==='products')setProducts(value);else setActivity(value);}catch(e){setListError((e as Error).message);}finally{listLocks.current.delete(kind);setListBusy(listLocks.current.size>0);}}
 const brandCounts=operations?.counts;const visibleRecent=activity.items;const visibleProducts=products?.items??recentProducts;
 const queue=[['pending_producer','Producer approvals','reviews/producer','pending'],['pending_product','Product approvals','reviews/product','pending'],['pending_buyer','Buyer approvals','reviews/buyer','pending'],['pending_inquiries','Unresolved inquiries','inquiries','unresolved']];
 return <Localized><div className="admin-dashboard overview-dashboard">
 <p className="overview-asof"><span>Data as of</span> <time dateTime={observed_at}>{time(observed_at)}</time></p>
 <section className="overview-kpis" aria-label="Operational indicators">{metrics.map(([key,label,target])=><button className="overview-kpi" key={key} aria-controls={target==='priority'?'overview-priority':undefined} onClick={()=>target==='priority'?document.getElementById('overview-priority')?.scrollIntoView({behavior:window.matchMedia('(prefers-reduced-motion: reduce)').matches?'instant':'smooth'}):select(target,key==='new_inquiries'?'received':undefined)}><span>{label}</span><strong>{number(brandCounts?.[key])}</strong></button>)}</section>
 <p className="overview-sales-note">Product and review totals include existing operational records. NOPAL product content is managed separately in code. Revenue and sales quantities are unavailable.</p>
 <div className="overview-charts"><figure className="overview-panel"><figcaption><h3>Registration approval status</h3><p>Current registrations, grouped by review state.</p></figcaption>{!approvalsReady?<p className="overview-unavailable">Approval breakdown was not queried.</p>:<><div className="overview-legend">{states.map(state=><span key={state}><i className={`overview-swatch ${state}`} aria-hidden="true"/>{statusNames[state]}</span>)}</div>{approvalData.map(({kind,label,data})=><section className="overview-bar-group" key={kind} aria-label={label}><header><h4>{label}</h4><span><span>Total</span> {number(states.reduce((a,s)=>a+data![s],0))}</span></header><div className="overview-bar-track" aria-hidden="true">{states.map(state=><span key={state} className={state} style={{width:`${data![state]/approvalMax*100}%`}}/>)}</div><dl>{states.map(state=><div key={state}><dt>{statusNames[state]}</dt><dd>{number(data![state])}</dd></div>)}</dl></section>)}{approvalData.every(({data})=>states.every(s=>data![s]===0))&&<p>No registrations recorded.</p>}</>}</figure>
 <figure className="overview-panel"><figcaption><h3>Inquiry handling status</h3><p>Current state of all saved inquiries.</p></figcaption>{!inquiryReady?<p className="overview-unavailable">Inquiry breakdown was not queried.</p>:<>{['received','reviewing','resolved'].map(state=><div className="overview-inquiry-bar" key={state}><div><span>{statusNames[state]}</span><strong>{number(inquiries![state])}</strong></div><div className="overview-bar-track" aria-hidden="true"><span className={state} style={{width:`${inquiries![state]/inquiryMax*100}%`}}/></div></div>)}{Object.values(inquiries!).every(v=>v===0)&&<p>No saved inquiries yet.</p>}<p className="overview-chart-note">Unresolved means received or reviewing.</p></>}</figure></div>
 {activityTrend?.length===7&&activityTrend.every(d=>valid(d.count)&&/^\d{4}-\d{2}-\d{2}$/.test(d.day))&&<figure className="overview-panel overview-trend"><figcaption><h3>Recent audited activity</h3><p>Last 7 UTC calendar days, including today. Audited actions, not new registrations.</p></figcaption><ol>{activityTrend.map(day=><li key={day.day}><strong>{number(day.count)}</strong><div className="overview-trend-track" aria-hidden="true"><span style={{height:`${day.count/Math.max(1,...activityTrend.map(d=>d.count))*100}%`}}/></div><time dateTime={day.day}>{new Intl.DateTimeFormat(locale,{month:'short',day:'numeric',timeZone:'UTC'}).format(new Date(day.day+'T00:00:00Z'))}</time></li>)}</ol>{activityTrend.every(d=>d.count===0)&&<p>No audited activity in this period.</p>}</figure>}
 <section className="overview-panel" aria-labelledby="overview-priority"><header className="overview-panel-title"><h3 id="overview-priority">Priority actions</h3><p>Open the queue to review and take action.</p></header><div className="overview-priorities">{queue.map(([key,label,target,status])=><div key={key}><div><strong id={`overview-queue-${key}`}>{label}</strong><span>{number(counts[key])} {valid(counts[key])&&<span>{counts[key]===0?'Up to date':'Awaiting action'}</span>}</span></div><button className="button-secondary" aria-labelledby={`overview-queue-${key} overview-open-${key}`} onClick={()=>select(target,status)}><span id={`overview-open-${key}`}>Open queue</span></button></div>)}</div></section>
 {listBusy&&<p role="status">Loading records…</p>}{listError&&<p role="alert">{listError}</p>}<section className="overview-panel" aria-labelledby="overview-recent" aria-busy={listBusy}><header className="overview-panel-title"><h3 id="overview-recent">Recent activity</h3><button className="button-secondary" onClick={()=>select('activity-log')}>Review audit records</button></header>{visibleRecent.length?<ol className="overview-activities">{visibleRecent.map(row=><li key={String(row.id)}><div><strong>{operationActions[String(row.action)]??actionNames[String(row.action)]??String(row.action).replaceAll('_',' ')}</strong><span className="overview-result" data-result={row.result}>{row.action==='buyer_unpublish'?'Withdrawn':statusNames[String(row.result)]??row.result??'Not provided'}</span></div><p><span>Target</span> <code data-no-translate>{String(row.target_id).slice(0,8)}</code> · <span>Actor</span> <code data-no-translate={Boolean(row.actor_id)}>{row.actor_id?String(row.actor_id).slice(0,8):'Automatic receipt'}</code></p><time dateTime={String(row.created_at)}>{time(row.created_at)}</time></li>)}</ol>:<p>No administrative changes recorded. Access grants are listed separately in Access History.</p>}<AdminPagination data={activity} busy={listBusy} onPage={page=>void loadList('activity-log',page)}/></section>
 <details className="overview-details" onToggle={e=>{if(e.currentTarget.open&&!products&&operations?.recentPagination)void loadList('products');}}><summary>More operational details</summary><div className="overview-secondary-links"><button className="button-secondary" onClick={()=>select('accounts','onboarding_required')}>Pending onboarding · {number(counts.pending_onboarding)}</button><button className="button-secondary" onClick={()=>select('products','draft')}>Draft products · {number(counts.drafts)}</button><button className="button-secondary" onClick={()=>select('grants')}>Access History · {number(counts.grant_events)}</button><button className="button-secondary" onClick={()=>select('readiness')}>{'Logistics & Readiness'}</button></div><h3>Recent product updates</h3>{visibleProducts===undefined?<p>Product updates were not queried.</p>:visibleProducts.length?<ul className="admin-activity-list">{visibleProducts.map(row=><li key={String(row.id)}><ProductImages images={parseProductImages(row.image_data||row.images)} name={String(row.name)} compact/><strong data-no-translate>{row.name}</strong><span>{row.state} · {row.approval_state}</span><time>{time(row.updated_at)}</time></li>)}</ul>:<p>No stored products have been registered or updated.</p>}{products&&<AdminPagination data={products} busy={listBusy} onPage={page=>void loadList('products',page)}/>}<button className="button-secondary" onClick={()=>select('products')}>Review products</button><h3>{'System health & data coverage'}</h3><dl className="admin-health-grid">{Object.entries(health).map(([key,value])=><div key={key}><dt>{key.replaceAll('_',' ')}</dt><dd>{value}</dd></div>)}</dl><p>Email delivery is not configured.</p></details>
 </div></Localized>;
}
