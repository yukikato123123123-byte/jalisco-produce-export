import {AdminPagination,type PageInfo} from './AdminPagination';
import {useCallback,useEffect,useRef,useState} from 'react';
import {accountRequest} from './AccountPages';
import {Localized,useLanguage} from '../localization';
import {WorkspaceIcon} from './WorkspaceIcon';
import './admin-notifications.css';
type Notification={id:number;user_id:string;name:string;email:string;role:string;created_at:string;state:string;read:number};
type Feed=PageInfo&{items:Notification[];unread_count:number;through:number};
export function AdminNotifications({onReview}:{onReview:(role:string,id:string)=>void}){
 const {language}=useLanguage(),[feed,setFeed]=useState<Feed|null>(null),[open,setOpen]=useState(false),[error,setError]=useState(''),[busy,setBusy]=useState(false),[loading,setLoading]=useState(false),[page,setPage]=useState(1);const trigger=useRef<HTMLButtonElement>(null),active=useRef(true),requestId=useRef(0);
 const load=useCallback(async(current=1)=>{const serial=++requestId.current;setLoading(true);try{const v=await accountRequest('/api/admin/notifications'+('?page='+current));if(active.current&&serial===requestId.current){setFeed(v);setError('');}}catch{if(active.current&&serial===requestId.current){setFeed(null);setError('Notifications could not be loaded. Refresh to retry.');}}finally{if(active.current&&serial===requestId.current)setLoading(false);}},[]);
 useEffect(()=>{const requests=requestId;active.current=true;const initial=setTimeout(()=>void load(open?page:1),0);const timer=setInterval(()=>{if(!document.hidden&&(!open||page===1))void load();},15000);return()=>{active.current=false;++requests.current;clearTimeout(initial);clearInterval(timer);};},[open,page,load]);
 async function mark(body:Record<string,number>){if(busy)return;setBusy(true);try{await accountRequest('/api/admin/notifications',body);await load(page);}catch{setFeed(null);setError('Notification update could not be confirmed. Refresh notifications before retrying.');}finally{setBusy(false);}}
 function close(){setOpen(false);trigger.current?.focus();}
 return <Localized><div className="admin-notifications" onKeyDown={e=>{if(e.key==='Escape')close();}}><button ref={trigger} className="button-secondary" aria-label="Registration notifications" aria-expanded={open} aria-controls={open?'admin-notification-panel':undefined} onClick={()=>{setOpen(!open);setPage(1);}}><WorkspaceIcon name="notifications"/><span className="admin-status">{Number.isFinite(feed?.unread_count)?feed!.unread_count:'—'}</span></button>{open&&<section id="admin-notification-panel" className="account-card admin-notification-panel" aria-label="Registration notifications" aria-busy={loading}><h3>New registrations</h3><p>Updates every 15 seconds while the console is open. Email confirmation and onboarding must be complete.</p><div><button className="button-secondary" disabled={loading} onClick={()=>void load(page)}>Refresh notifications</button><button className="button-secondary" disabled={busy||!feed?.unread_count} onClick={()=>void mark({through:feed?.through||0})}>Mark all as read</button><button className="button-secondary" onClick={close}>Close notifications</button></div>{error&&<p className="account-error" role="alert">{error}</p>}{!feed&&!error&&<p role="status">Loading notifications…</p>}{feed?.items?.length===0&&<p>No registration notifications yet.</p>}<ul className="admin-notification-list">{feed?.items?.map(n=><li key={n.id}><strong data-no-translate>{n.name}</strong><span>{n.role==='producer'?'Producer':'Buyer'}</span><span data-no-translate>{n.email}</span><time dateTime={n.created_at}>{new Intl.DateTimeFormat(language==='es'?'es-MX':'en-US',{dateStyle:'medium',timeStyle:'short'}).format(new Date(n.created_at))}</time><span>Approval state: <span>{n.state}</span></span><span>{n.read?'Read':'Unread'}</span><div><button className="button-secondary" onClick={()=>{onReview(n.role,n.user_id);close();}}>Review registration</button>{!n.read&&<button className="button-secondary" disabled={busy} onClick={()=>void mark({id:n.id})}>Mark as read</button>}</div></li>)}</ul>{feed&&<AdminPagination data={feed} busy={busy||loading} onPage={setPage}/>}</section>}</div></Localized>;
}
