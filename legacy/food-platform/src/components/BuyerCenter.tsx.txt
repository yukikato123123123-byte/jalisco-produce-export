import {Localized} from '../localization';
import { BuyerPublication } from './BuyerPublication';
import type { BuyerPublicationRecord } from './BuyerPublication';
import { useEffect, useRef, useState } from 'react';
import { accountRequest } from './AccountPages';
import { ProduceBrowser, ProduceCard } from './ExploreProduce';
import type { PublicItem } from './ExploreProduce';
type BuyerData={profile:{company:string;country:string;region:string;bio:string};publication?:BuyerPublicationRecord|null;savedItems:PublicItem[];page:number;hasMore:boolean};
export function BuyerCenter(){
  const [data,setData]=useState<BuyerData|null>(null),[error,setError]=useState(''),[message,setMessage]=useState(''),[busy,setBusy]=useState(false);const lock=useRef(false);
  const [savedQuery,setSavedQuery]=useState('');
  useEffect(()=>{let active=true;accountRequest('/api/buyer/dashboard').then(v=>{if(active)setData(v);}).catch(e=>{if(active)setError(e.message);});return()=>{active=false;};},[]);
  async function request(path:string,body?:Record<string,unknown>){if(lock.current)return;lock.current=true;setBusy(true);setError('');setMessage('');try{setData(await accountRequest(path,body));setMessage('Changes saved.');}catch(e){setError(e instanceof Error?e.message:'Unable to save.');}finally{lock.current=false;setBusy(false);}}
  return <Localized><section aria-label="Buyer Center"><nav className="workspace-nav" aria-label="Buyer workspace sections"><a href="#buyer-profile">Company profile</a><a href="#public-buyer-profile">Public directory profile</a><a href="#saved-produce">Saved produce</a><a href="#find-suppliers">Find suppliers</a></nav><h3 id="buyer-profile">Buyer company profile</h3><p>Tell us a little about your business. Your buyer profile stays private to your account.</p>{error&&<p role="alert" className="account-error">{error}</p>}{message&&<p role="status" className="account-success">{message}</p>}{!data?!error&&<p role="status">Loading Buyer Center…</p>:<><form onSubmit={e=>{e.preventDefault();void request('/api/buyer/profile',Object.fromEntries(new FormData(e.currentTarget)));}}><fieldset className="account-fields workspace-fields" disabled={busy}>{(['company','country','region','bio'] as const).map(k=><label className={k==='bio'?'field-wide':undefined} key={k}>{({company:'Buyer company',country:'Buyer country',region:'Buyer region',bio:'Buyer introduction'})[k]}{k==='bio'?<textarea name={k} defaultValue={data.profile[k]} maxLength={2000} rows={4}/>:<input name={k} defaultValue={data.profile[k]} required={k==='company'||k==='country'} maxLength={120}/>}</label>)}<button>Save buyer profile</button></fieldset></form><BuyerPublication publication={data.publication} busy={busy} onSave={values=>void request('/api/buyer/publication',values)}/><h3 id="saved-produce">Saved produce</h3><label>Search saved produce on this page<input value={savedQuery} onChange={e=>setSavedQuery(e.target.value)} placeholder="Produce or supplier"/></label><p>Manage your sourcing shortlist. Items become unavailable here when producers stop publishing them.</p>{!data.savedItems.length&&<p>No public items saved yet.</p>}<div className="produce-grid">{data.savedItems.length>0&&!data.savedItems.some(item=>(item.name+' '+item.company).toLowerCase().includes(savedQuery.toLowerCase()))&&<p role="status">No saved items match your search on this page.</p>}{data.savedItems.filter(item=>(item.name+' '+item.company).toLowerCase().includes(savedQuery.toLowerCase())).map(item=><ProduceCard key={item.id} item={item} action={<button className="button-secondary" disabled={busy} onClick={()=>void request('/api/buyer/saved-items',{id:item.id,action:'remove'})}>Remove {<span data-no-translate>{item.name}</span>}</button>}/>)}</div><div className="produce-pagination">{data.page>1&&<button disabled={busy} onClick={()=>void request(`/api/buyer/dashboard?page=${data.page-1}`)}>Previous saved page</button>}{data.hasMore&&<button disabled={busy} onClick={()=>void request(`/api/buyer/dashboard?page=${data.page+1}`)}>Next saved page</button>}</div><h3 id="find-suppliers">Find produce and suppliers</h3><a className="text-link" href="/producers">Browse the producer directory </a><ProduceBrowser busy={busy} onSave={id=>void request('/api/buyer/saved-items',{id,action:'save'})}/><p>Orders, payments and logistics tracking are planned features and are not available.</p></>}
  </section></Localized>;
}
