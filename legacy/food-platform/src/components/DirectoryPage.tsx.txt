import {ProductImages} from './ProductImages';
import type {ProductImage} from '../productImages';
import {Localized} from '../localization';
import { BuyerDirectoryEntry } from './BuyerDirectoryEntry';
import type { PublicBuyer } from './BuyerDirectoryEntry';
import { useEffect, useState } from 'react';
import type { FormEvent } from 'react';
import { accountRequest } from './AccountPages';
import { SiteHeader, SiteFooter } from './SiteChrome';
import { FarmStory } from './FarmStory';
type Producer={id:string;company:string;country:string;region:string;bio:string;business_email:string;business_phone:string;items:{images?:ProductImage[];id:string;name:string;category:string;quantity:number;unit:string;availability:string}[]};
type Result={items:(Producer & Partial<PublicBuyer>)[];page:number;hasMore:boolean;message?:string};
export function DirectoryPage(){
  const buyer=window.location.pathname.startsWith('/buyers');
  const section=buyer?'buyers':'producers';
  const detail=window.location.pathname.split('/')[2]||'';
  const editorial=!buyer&&detail==='field-notes';
  const [params,setParams]=useState(()=>{const p=new URLSearchParams(window.location.search);if(detail&&!editorial){try{p.set('id',decodeURIComponent(detail));}catch{p.set('id',detail);}};return p;});
  const [result,setResult]=useState<Result|null>(null),[error,setError]=useState('');
  useEffect(()=>{if(editorial)return;let active=true;accountRequest(`/api/public/${section}?${params}`).then(v=>{if(active)setResult(v);}).catch(e=>{if(active)setError(e.message);});return()=>{active=false;};},[section,params,editorial]);
  function navigate(next:URLSearchParams){setResult(null);setError('');setParams(next);window.history.replaceState(null,'',`/${section}${next.size?'?'+next:''}`);}
  function search(e:FormEvent<HTMLFormElement>){e.preventDefault();const p=new URLSearchParams();new FormData(e.currentTarget).forEach((v,k)=>{if(String(v).trim())p.set(k,String(v).trim());});navigate(p);}
  return <Localized><div className={`account-page directory-page ${buyer?'buyer-directory':'producer-directory'}`}><SiteHeader/><main className="directory-main">
    <header className="directory-heading"><div><p className="eyebrow">{buyer?'MARKETS · RELATIONSHIPS · OPPORTUNITY':'PEOPLE · PLACE · PRODUCE'}</p><h1>{editorial?'From the fields of Jalisco':detail?(buyer?'Buyer profile':'Meet the producer'):(buyer?'Find your next market.':'Meet the people behind the produce.')}</h1></div><p>{buyer?'Connect around shared sourcing interests. Only information approved for public discovery belongs here.':'Explore producers through the produce they have chosen to publish. Start a conversation about origin, availability and the next harvest.'}</p></header>
    {editorial?<><FarmStory full/><p className="directory-note">Photographs supplied by Edgar. The farm feature is editorial; it is not a verified supplier listing or a live inventory record.</p><a className="text-link" href="/producers"> Back to producers</a></>:<>
      {!detail&&!buyer&&<a className="farm-feature" href="/producers/field-notes"><img src="/images/el-grullo-fields.jpeg" alt="Nopal fields in the farm photographs supplied by Edgar"/><div><p className="eyebrow">EDITORIAL · NOT A DIRECTORY RECORD</p><h2>A closer look at El Grullo.</h2><p>See the origin through Edgar’s farm photographs.</p><span className="text-link">Discover the farm </span></div></a>}
      <section className="directory-content account-card" aria-label={buyer?'Buyer directory':'Producer directory'}>
        {!detail&&<form className="directory-filters" onSubmit={search} key={params.toString()}><label>{buyer?'Company or sourcing interest':'Producer or produce'}<input name="q" maxLength={120} defaultValue={params.get('q')||''}/></label><label>{buyer?'Country or market':'Country'}<input name="country" maxLength={120} defaultValue={params.get('country')||''}/></label>{buyer&&<><label>Target market<input name="market" maxLength={120} defaultValue={params.get('market')||''}/></label><label>Produce interest<input name="interest" maxLength={120} defaultValue={params.get('interest')||''}/></label></>}{!buyer&&<label>Region<input name="region" maxLength={120} defaultValue={params.get('region')||''}/></label>}<button>{buyer?'Search buyers':'Search producers'}</button><button className="button-secondary" type="button" onClick={()=>navigate(new URLSearchParams())}>Clear filters</button></form>}
        {error&&<div role="alert" className="account-error"><p>{error}</p><button className="button-secondary" onClick={()=>{setError('');setResult(null);setParams(new URLSearchParams(params));}}>Try again</button></div>}
        {!result&&!error&&<p role="status">Loading {section}…</p>}
        {result&&<><p role="status">{result.items.length?`${result.items.length} public ${buyer?(result.items.length===1?'buyer':'buyers'):(result.items.length===1?'producer':'producers')} on this page`:detail?'This public profile is unavailable.':buyer?'No public buyer profiles are available.':'No published producers match these filters.'}</p>{!result.items.length&&<div className="directory-empty"><span aria-hidden="true">{buyer?'':''}</span><h2>{buyer?'Good connections start with permission.':'The next harvest starts here.'}</h2><p>{buyer?(result.message||'Only explicitly published buyer profiles appear here. Private company profiles stay private.'):'A registered account appears here only after its producer publishes a product. Drafts and private profiles stay private. Try another region or product.'}</p><a className="platform-button" href={buyer?'/signup':'/explore'}>{buyer?'Create a buyer account':'Explore Produce '}</a>{buyer&&<p>Have a producer account? <a className="text-link" href="/producer/dashboard">Open My Workspace </a></p>}</div>}
          <div className="producer-directory-grid">{result.items.map(p=>buyer?<BuyerDirectoryEntry key={p.id} buyer={p as PublicBuyer} detail={!!detail}/>:<article className="producer-directory-entry" key={p.id}><p className="eyebrow" data-no-translate>{[p.region,p.country].filter(Boolean).join(' · ')||'Location not provided'}</p><h2 data-no-translate>{p.company}</h2><p data-no-translate={Boolean(p.bio)}>{p.bio||'Explore this producer’s published harvest.'}</p><p className="publication-note">Producer reported · No verification implied</p>{detail?<><h3>Published produce & supply</h3><div className="directory-supply">{p.items.map(i=><article key={i.id}><ProductImages images={i.images} name={i.name}/><h4><a href={`/explore?id=${encodeURIComponent(i.id)}`}>{i.name} </a></h4><p>{i.quantity} {i.unit} · {i.category}</p><p>{i.availability||'Contact producer to confirm availability'}</p></article>)}</div><h3>Start a conversation</h3>{p.business_email&&<p>Public business email: <a className="text-link" href={`mailto:${p.business_email}`} data-no-translate>{p.business_email}</a></p>}{p.business_phone&&<p>Public business phone: {<span data-no-translate>{p.business_phone}</span>}</p>}{!p.business_email&&!p.business_phone&&<p>This producer has not published business contact details. No message has been sent.</p>}<a className="text-link" href="/producers"> All producers</a></>:<><ProductImages images={p.items[0]?.images} name={p.items[0]?.name||p.company}/><p>{p.items.map(i=>i.name).join(' · ')}</p><a className="text-link" href={`/producers/${encodeURIComponent(p.id)}`}>View producer & public supply </a></>}</article>)}</div>
          <div className="produce-pagination">{result.page>1&&<button onClick={()=>{const p=new URLSearchParams(params);p.set('page',String(result.page-1));navigate(p);}}>Previous page</button>}{result.hasMore&&<button onClick={()=>{const p=new URLSearchParams(params);p.set('page',String(result.page+1));navigate(p);}}>Next page</button>}</div>
        </>}
        {detail&&!result?.items.length&&<a className="text-link" href={`/${section}`}> Back to {section}</a>}
      </section><p className="directory-note">{buyer?'Private contacts, saved sourcing lists and commercial activity are never part of this directory.':'Publication does not verify a supplier, certify export readiness or guarantee availability. Confirm quantities and terms directly with the producer.'}</p>
    </>}
  </main><SiteFooter/></div></Localized>;
}
