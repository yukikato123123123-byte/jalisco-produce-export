import {ProductImages} from './ProductImages';
import type {ProductImage} from '../productImages';
import {useLanguage,Localized} from '../localization';
import { useEffect, useState } from 'react';
import type { FormEvent } from 'react';
import { accountRequest } from './AccountPages';
import { SiteHeader, SiteFooter } from './SiteChrome';
export type PublicItem = {images?:ProductImage[];id:string;name:string;description:string;category:string;quantity:number;unit:string;price:number|null;currency:string;availability:string;company:string;country:string;region:string;bio:string;business_email:string;business_phone:string;producer_id?:string};
export function ProduceCard({item,action}:{item:PublicItem;action?: React.ReactNode}) {
  const {language}=useLanguage();const locale=language==='es'?'es-MX':'en-US';
  return <Localized><article className="produce-card"><ProductImages images={item.images} name={item.name}/><div className="produce-card-marker" aria-hidden="true"> <span>FROM THE PRODUCER</span></div><p className="eyebrow">{item.category}</p><h3 data-no-translate>{item.name}</h3><p data-no-translate>{item.description}</p><p><strong data-no-translate>{item.company}</strong> · {<span data-no-translate>{[item.region,item.country].filter(Boolean).join(', ')}</span>}</p><p>{item.quantity.toLocaleString(locale,{maximumFractionDigits:20})} {item.unit}{item.price!==null?` · ${item.price.toLocaleString(locale,{maximumFractionDigits:20})} ${item.currency}`:''}</p><details open={window.location.pathname==='/explore'&&new URLSearchParams(window.location.search).get('id')===item.id}><summary>Producer and supply details</summary><ProductImages images={item.images} name={item.name} gallery/><p data-no-translate={Boolean(item.bio)}>{item.bio||'No producer introduction provided.'}</p><p>Availability: {item.availability?<span data-no-translate>{item.availability}</span>:'Contact producer to confirm'}</p>{item.business_email&&<p>Business email: <a className="text-link" href={`mailto:${item.business_email}`} data-no-translate>{item.business_email}</a></p>}{item.business_phone&&<p>Business phone: {<span data-no-translate>{item.business_phone}</span>}</p>}{!item.business_email&&!item.business_phone&&<p>Business contact details are not public.</p>}<a className="text-link" href={`/producers/${encodeURIComponent(item.producer_id||item.id)}`}>Meet this producer </a><br/><a className="text-link" href={`/explore?id=${encodeURIComponent(item.id)}`}>Link to this public item </a></details>{action}</article></Localized>;
}
export function ProduceBrowser({onSave,busy=false}:{onSave?:(id:string)=>void;busy?:boolean}) {
  const [params,setParams]=useState(()=>new URLSearchParams(window.location.pathname==='/explore'?window.location.search:''));
  const [result,setResult]=useState<{items:PublicItem[];page:number;hasMore:boolean}|null>(null),[error,setError]=useState('');
  useEffect(()=>{let active=true;accountRequest(`/api/public/produce?${params}`).then(v=>{if(active){setResult(v);setError('');}}).catch(e=>{if(active)setError(e.message);});return()=>{active=false;};},[params]);
  function navigate(next:URLSearchParams){setResult(null);setError('');setParams(next);if(window.location.pathname==='/explore')window.history.replaceState(null,'',`/explore${next.size?'?'+next:''}`);}
  function search(e:FormEvent<HTMLFormElement>){e.preventDefault();const next=new URLSearchParams();new FormData(e.currentTarget).forEach((v,k)=>{if(String(v).trim())next.set(k,String(v).trim());});navigate(next);}
  return <Localized><section aria-label="Public produce search"><form key={params.toString()} onSubmit={search} className="produce-filters">{[['q','Produce or producer'],['country','Country'],['region','Region'],['category','Variety or category']].map(([name,label])=><label key={name}>{label}<input name={name} maxLength={120} defaultValue={params.get(name)||''}/></label>)}<button>Search produce</button><button className="button-secondary" type="button" onClick={e=>{e.currentTarget.form?.reset();navigate(new URLSearchParams());}}>Reset search</button></form>{error&&<div role="alert" className="account-error"><p>{error}</p><button className="button-secondary" onClick={()=>navigate(new URLSearchParams(params))}>Retry search</button></div>}{!result&&!error&&<p role="status">Loading public produce…</p>}{result&&<><p role="status">{result.items.length?`${result.items.length} public items on this page`:'No public produce matches your search. Try other filters or reset search.'}</p><div className="produce-grid">{result.items.map(item=><ProduceCard key={item.id} item={item} action={onSave&&<button disabled={busy} onClick={()=>onSave(item.id)}>Save {<span data-no-translate>{item.name}</span>}</button>}/>)}</div><div className="produce-pagination">{result.page>1&&<button onClick={()=>{const next=new URLSearchParams(params);next.set('page',String(result.page-1));navigate(next);}}>Previous page</button>}{result.hasMore&&<button onClick={()=>{const next=new URLSearchParams(params);next.set('page',String(result.page+1));navigate(next);}}>Next page</button>}</div></>}</section></Localized>;
}
export function ExploreProduce(){return <Localized><div className="account-page"><SiteHeader/><main className="explore-layout account-card"><p className="eyebrow">PUBLIC PRODUCE DIRECTORY</p><h1>Explore Produce</h1><p>Browse produce published by producers. Listings and available supply are producer reported; publication does not verify a supplier or guarantee an offer.</p><div className="explore-intro"><div><p>From origin to opportunity.
