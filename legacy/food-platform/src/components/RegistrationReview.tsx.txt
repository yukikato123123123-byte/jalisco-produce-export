import {AdminPagination,type PageInfo} from './AdminPagination';
import {useRef,useState} from 'react';
import type {FormEvent} from 'react';
import {accountRequest} from './AccountPages';
import {Localized,useLanguage} from '../localization';
type Row=Record<string,string|number|null>;
export function RegistrationReview({kind,row,onSaved}:{kind:string;row:Row;onSaved:(state:string,eventId:string|null,revision:number)=>void}){
 const {language}=useLanguage(),[busy,setBusy]=useState(false),[error,setError]=useState(''),[notice,setNotice]=useState(''),[history,setHistory]=useState<Row[]|null>(null);const [pagination,setPagination]=useState<PageInfo>({});const lock=useRef(false),retry=useRef<{key:string;id:string}|null>(null);
 async function save(e:FormEvent<HTMLFormElement>){e.preventDefault();if(lock.current)return;const after=(e.nativeEvent as SubmitEvent).submitter?.getAttribute('value');if(!after)return;const form=e.currentTarget,reason=String(new FormData(form).get('reason')||'');const payload={revision:row.revision,before_state:row.state,after_state:after,expected_event_id:row.last_event_id,reason};const key=JSON.stringify(payload);if(retry.current?.key!==key)retry.current={key,id:crypto.randomUUID()};const id=retry.current.id;lock.current=true;setBusy(true);setError('');setNotice('');try{await accountRequest(`/api/admin/reviews/${kind}/${encodeURIComponent(String(row.id))}`,{...payload,id});const detail=await accountRequest(`/api/admin/reviews/${kind}/${encodeURIComponent(String(row.id))}?page=999999`);setHistory(detail.history);setPagination(detail.historyPagination??{});onSaved(detail.review.state,detail.review.last_event_id,detail.review.revision);setNotice('Review decision saved with an audit record.');form.reset();retry.current=null;}catch(e){setError((e as Error).message);}finally{setBusy(false);lock.current=false;}}
 async function loadHistory(page=1){if(lock.current)return;lock.current=true;setBusy(true);setError('');try{const v=await accountRequest(`/api/admin/reviews/${kind}/${encodeURIComponent(String(row.id))}?page=${page}`);setHistory(v.history);setPagination(v.historyPagination??{});}catch(e){setError((e as Error).message);}finally{lock.current=false;setBusy(false);}}
 return <Localized><div className="registration-review"><form onSubmit={save}><p>Approval does not publish a private record or grant account privileges. Content changes require another review.</p><label>Review reason<textarea name="reason" maxLength={500} required disabled={busy}/></label><div className="review-decisions"><button type="submit" name="decision" value="approved" disabled={busy||row.state==='approved'}>Approve registration</button><button type="submit" name="decision" value="rejected" className="button-danger" disabled={busy||row.state==='rejected'}>Reject registration</button></div></form>{error&&<p role="alert">{error}</p>}{notice&&<p className="admin-success" role="status">{notice}</p>}<button type="button" className="button-secondary" disabled={busy} onClick={()=>void loadHistory()}>View review history</button>{history&&<ol className="inquiry-timeline">{history.length===0&&<li>No review decisions recorded.</li>}{history.map(event=><li key={String(event.id)}><strong>{event.state}</strong><p><span>Revision</span>: {event.revision}</p><p><span>Recorded by</span>: <span data-no-translate>{event.actor_id}</span></p><time>{new Intl.DateTimeFormat(language==='es'?'es-MX':'en-US',{dateStyle:'medium',timeStyle:'short'}).format(new Date(String(event.created_at)))}</time><p data-no-translate>{event.reason}</p></li>)}</ol>}{history&&<AdminPagination data={pagination} busy={busy} onPage={page=>void loadHistory(page)}/>}</div></Localized>;
}
