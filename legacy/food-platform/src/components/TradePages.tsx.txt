import {ProductImages} from './ProductImages';
import {Localized} from '../localization';
import { useEffect, useState } from 'react';
import { SiteHeader, SiteFooter } from './SiteChrome';
import { accountRequest } from './AccountPages';
import type { PublicItem } from './ExploreProduce';
import { MarketMap } from './MarketMap';
type Workspace={profile:Record<string,string>;summary:{profileCompletion:number;public:number;total:number}};
export function TradePage(){
 const section=window.location.pathname.slice(1);
 const [items,setItems]=useState<PublicItem[]|null>(null),[error,setError]=useState(''),[workspace,setWorkspace]=useState<Workspace|null>(null),[notice,setNotice]=useState(''),[reload,setReload]=useState(0);
 useEffect(()=>{let active=true;accountRequest('/api/public/produce').then(v=>{if(active)setItems(v.items);}).catch(e=>{if(active)setError(e.message);});if(section==='readiness')accountRequest('/api/auth/session').then(async session=>{if(session.recovery||session.profile?.role!=='producer'){if(active)setNotice('Open a producer workspace to review your stored profile and product completeness.');return;}const value=await accountRequest('/api/producer/dashboard');if(active)setWorkspace(value);}).catch(e=>{if(active)setNotice(e.message);});return()=>{active=false;};},[section,reload]);
 const title=section==='logistics'?'From origin to destination.':section==='commercial'?'Start with the details.':'Prepare the next step.';
 return <Localized><div className="account-page"><SiteHeader/><main className="directory-main"><header className="directory-heading"><div><p className="eyebrow">{section==='logistics'?'ORIGIN · HANDLING · DELIVERY':section==='commercial'?'SPECIFICATIONS · REQUIREMENTS · CONVERSATION':'COMPLETENESS · DOCUMENTATION · PREPARATION'}</p><h1>{title}</h1></div><p>{section==='logistics'?'A planning workspace for the questions that connect the harvest to its next market.':section==='commercial'?'Review producer-published details, then confirm specifications and terms directly.':'Separate what has been provided from what still needs confirmation. Completeness does not establish certification or legal compliance.'}</p></header>
 <section className="account-card"><h2>{section==='logistics'?'Logistics':section==='commercial'?'Commercial information':'Readiness checklist'}</h2>
 {section==='logistics'&&<><div className="home-paths">{[['01','At origin','Confirm the farm, harvest date, collection point and loading arrangements.'],['02','Packing & handling','Confirm packaging, net weights, pallet details, temperature and handling requirements with the producer.'],['03','Delivery planning','Agree destination, carrier, customs responsibilities and delivery conditions before booking.']].map(([n,h,p])=><article key={n}><span>{n}</span><h3>{h}</h3><p>{p}</p><span className="admin-status">Pending confirmation</span></article>)}</div><p>No carrier, shipment, transit time, rate or service availability is stored. The map below is illustrative and does not establish a working transport service.</p></>}
 {section==='commercial'&&<><p>Quantities, availability and optional asking prices below are self-reported public listing fields. They are not negotiated quotations, inventory guarantees or transactions.</p>{error&&<p role="alert" className="account-error">{error}</p>}{!items&&!error&&<p role="status">Loading published specifications…</p>}{items&&!items.length&&<p role="status">No public listings available. No prices or supply have been inferred.</p>}{!!items?.length&&<div className="admin-table-wrap" tabIndex={0} aria-label="Published commercial details"><table className="admin-table"><thead><tr><th>Product / producer</th><th>Reported quantity</th><th>Asking price</th><th>Availability</th></tr></thead><tbody>{items.map(i=><tr key={i.id}><td><ProductImages images={i.images} name={i.name} compact/><a href={`/explore?id=${encodeURIComponent(i.id)}`}>{i.name}</a><br/>{i.company}<p>{i.description||'Specifications not provided'}</p></td><td>{i.quantity} {i.unit}</td><td>{i.price===null?'Not provided':`${i.price} ${i.currency}`}</td><td>{i.availability||'Confirm with producer'}</td></tr>)}</tbody></table><p>First page of published records. <a href="/explore">Search all produce </a></p></div>}<dl><dt>Packaging specifications</dt><dd>Not stored on current listings. Confirm directly.</dd><dt>Incoterm, named place & delivery expectations</dt><dd>To be agreed. No negotiated terms are stored.</dd><dt>Inquiry workflow</dt><dd>Use the public contact details a producer chose to publish. Signed-in users can submit and review saved inquiries.</dd></dl><a className="text-link" href="/inquiries">Submit an inquiry </a></>}
 {section==='readiness'&&<><p>{workspace?'Based on your current stored producer workspace. All supplied information remains self-reported.':'Sign in as a producer for stored profile and publication checks. Documentation and logistics are unverified until supporting records exist.'}</p>{notice&&<p role="status">{notice}</p>}<div className="directory-supply">{[
 ['Producer profile',workspace?(workspace.summary.profileCompletion===100?'Complete · self-reported':`Incomplete · ${workspace.summary.profileCompletion}% of workspace fields provided`):'Unverified · sign in to inspect'],
 ['Published produce',workspace?`${workspace.summary.public} public / ${workspace.summary.total} stored products`:'Unverified · sign in to inspect'],['Packaging','Incomplete · no packaging specification records'],['Quality documentation','Unverified · no supporting document records'],['Export documents','Unverified · no supporting document records'],['Logistics preparation','Pending · no shipment or carrier records']
 ].map(([h,s])=><article key={h}><h3>{h}</h3><p>{s}</p></article>)}</div><a className="text-link" href="/producer/dashboard">Open My Workspace </a></>}
 {(error||section==='readiness')&&<button className="button-secondary" onClick={()=>{setError('');setNotice('');setReload(v=>v+1);}}>Refresh stored information</button>}
 <nav className="workspace-nav" aria-label="Next steps"><a href="/explore">Find produce </a><a href="/producers">Contact a producer </a><a href="/readiness">Review readiness </a></nav></section>
 {section==='logistics'&&<section className="business-section map-presentation" aria-label="Illustrative logistics map"><MarketMap active/></section>}
 </main><SiteFooter/></div></Localized>;
}
