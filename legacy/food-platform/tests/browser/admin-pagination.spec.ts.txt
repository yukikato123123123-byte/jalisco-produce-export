import {test,expect} from '@playwright/test';
for(const width of [375,768,1440])test(`admin server pages, filters, details and EN/ES at ${width}px`,async({page})=>{
 await page.setViewportSize({width,height:1000});let status='';let count=21;
 await page.route('**/api/admin/**',route=>{const url=new URL(route.request().url()),path=url.pathname,current=Math.min(Number(url.searchParams.get('page')||1),Math.max(1,Math.ceil(count/10)));if(path.endsWith('/notifications'))return route.fulfill({json:{items:[],unread_count:0,through:0,total:0,page:1,totalPages:1,pageSize:10}});if(path.endsWith('/products')){if(url.searchParams.get('q')){count=1;status='filtered';}else count=21;const n=Math.min(Number(url.searchParams.get('page')||1),Math.max(1,Math.ceil(count/10)));return route.fulfill({json:{items:Array.from({length:Math.min(10,count-(n-1)*10)},(_,i)=>({id:'product-'+((n-1)*10+i),name:status||'Synthetic product '+((n-1)*10+i),state:'private',revision:1})),page:n,pageSize:10,total:count,totalPages:Math.ceil(count/10),hasMore:n<Math.ceil(count/10)}});}return route.fulfill({json:{items:[],page:current,pageSize:10,total:0,totalPages:1}});});
 await page.goto('/admin?section=products');const pager=page.locator('.admin-content>.admin-pagination');await expect(pager).toContainText('Showing 1–10 of 21');await expect(pager.getByRole('button',{name:'Previous',exact:true})).toBeDisabled();await expect(page.locator('.admin-table>tbody>tr')).toHaveCount(10);
 await pager.getByRole('button',{name:'Next',exact:true}).click();await expect(pager).toContainText('Showing 11–20 of 21');await page.getByRole('button',{name:'View details',exact:true}).first().click();await expect(page.locator('#record-detail-panel')).toBeVisible();await pager.getByRole('button',{name:'Go to page 3',exact:true}).click();await expect(pager).toContainText('Showing 21–21 of 21');await expect(pager.getByRole('button',{name:'Next',exact:true})).toBeDisabled();await expect(page.locator('#record-detail-panel')).toHaveCount(0);await pager.getByRole('button',{name:'Previous',exact:true}).click();await expect(pager).toContainText('Showing 11–20 of 21');await page.getByLabel('Search records', {exact:true}).fill('one');await page.getByRole('button',{name:'Search',exact:true}).click();await expect(pager).toContainText('Showing 1–1 of 1');await expect(pager.locator('nav')).toHaveCount(0);
 await page.getByRole('combobox',{name:'Language / Idioma',exact:true}).selectOption('es');await expect(pager).toContainText('Mostrando 1–1 de 1');await page.getByRole('button',{name:'Limpiar',exact:true}).click();await expect(pager).toContainText('Mostrando 1–10 de 21');await expect(pager.getByRole('button',{name:'Siguiente',exact:true})).toBeVisible();expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth)).toBe(true);for(const button of await pager.locator('button').all()){const box=await button.boundingBox();expect(box!.x).toBeGreaterThanOrEqual(0);expect(box!.x+box!.width).toBeLessThanOrEqual(width);}
});
test('out of range response renders clamped page and failures remain errors rather than empty data',async({page})=>{
 let calls=0;await page.route('**/api/admin/**',route=>{const path=new URL(route.request().url()).pathname;if(path.endsWith('/notifications'))return route.fulfill({json:{items:[],unread_count:0,through:0}});if(path.endsWith('/launch-list')){calls++;if(calls===3)return route.fulfill({status:503,json:{error:'Brand operations data is unavailable.'}});return route.fulfill({json:{items:[{id:'saved',email:'synthetic@example.test',language:'en',created_at:'2026-10-01',state:'requested'}],page:1,pageSize:10,total:calls===1?11:1,totalPages:calls===1?2:1}});}return route.fulfill({json:{}});});await page.goto('/admin?section=launch-list');await page.getByRole('button',{name:'Next',exact:true}).click();await expect(page.locator('.admin-pagination')).toContainText('Showing 1–1 of 1');await page.getByRole('button',{name:'Refresh records',exact:true}).click();await expect(page.getByRole('alert')).toBeVisible();await expect(page.locator('.admin-pagination')).toHaveCount(0);
});
test('feedback page two retains review and notes while refreshing filtered totals',async({page})=>{
 let changed=false;const history=Array.from({length:11},(_,i)=>({id:'event-'+i,state:'received',note:'History '+i,created_at:'2026-10-01'}));
 await page.route('**/api/admin/**',route=>{const url=new URL(route.request().url()),path=url.pathname;if(path.endsWith('/notifications'))return route.fulfill({json:{items:[],unread_count:0,through:0}});if(path.includes('/customer-feedback/')){if(route.request().method()==='POST'){const body=route.request().postDataJSON();changed=true;history.push({id:body.id,state:'reviewing',note:body.note,created_at:'2026-10-01'});return route.fulfill({json:{ok:true}});}const current=Math.min(Number(url.searchParams.get('historyPage')||1),2);return route.fulfill({json:{item:{id:'feedback-10',category:'idea',message:'Synthetic full feedback',state:changed?'reviewing':'received',last_event_id:history.at(-1)!.id,created_at:'2026-10-01',reply_consent:0},history:history.slice((current-1)*10,current*10),historyPagination:{page:current,pageSize:10,total:history.length,totalPages:2},can_manage:true}});}const current=Number(url.searchParams.get('page')||1),total=changed?20:21;return route.fulfill({json:{items:Array.from({length:Math.min(10,total-(current-1)*10)},(_,i)=>({id:'feedback-'+((current-1)*10+i),category:'idea',excerpt:'Synthetic opinion',state:'received',created_at:'2026-10-01',reply_consent:0})),page:current,pageSize:10,total,totalPages:Math.ceil(total/10)}});});
 await page.goto('/admin?section=customer-feedback');await page.getByRole('combobox',{name:'Feedback status',exact:true}).selectOption('received');await page.getByRole('button',{name:'Search',exact:true}).click();await page.locator('.feedback-inbox>.admin-pagination').getByRole('button',{name:'Next',exact:true}).click();await page.getByRole('button',{name:'Review feedback',exact:true}).first().click();await expect(page.locator('.feedback-history>li')).toHaveCount(10);await page.locator('.feedback-detail .admin-pagination').getByRole('button',{name:'Next',exact:true}).click();await expect(page.locator('.feedback-history>li')).toHaveCount(1);await page.getByLabel('Processing status',{exact:true}).selectOption('reviewing');await page.getByLabel('Internal review note',{exact:true}).fill('Saved from page two');await page.getByRole('button',{name:'Save internal review',exact:true}).click();await expect(page.locator('.feedback-history')).toContainText('Saved from page two');await expect(page.locator('.feedback-inbox>.admin-pagination')).toContainText('Showing 11–20 of 20');await expect(page.locator('.feedback-detail .admin-pagination')).toContainText('Showing 11–12 of 12');
});
