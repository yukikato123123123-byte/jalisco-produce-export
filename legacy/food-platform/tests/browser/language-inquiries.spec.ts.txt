import {test,expect} from '@playwright/test';
const key='jalisco-produce-export.language';
const paths=['/','/explore','/producers','/buyers','/login','/signup','/forgot-password','/verify-email','/reset-password','/inquiries','/admin','/commercial','/readiness','/logistics'];
for(const lang of ['en','es'])test(`global ${lang} selector persists across all main routes and reloads`,async({page})=>{
 await page.addInitScript(({key,lang})=>{if(!localStorage.getItem(key))localStorage.setItem(key,lang);},{key,lang});
 await page.route('**/api/**',r=>{const u=new URL(r.request().url());if(u.pathname==='/api/auth/session'||u.pathname==='/api/inquiries'||u.pathname.startsWith('/api/admin/'))return r.fulfill({status:401,json:{error:'Please sign in to continue.'}});return r.fulfill({json:{items:[],page:1,hasMore:false}});});
 for(const path of paths){await page.goto(path);if(path==='/'||path==='/admin')await expect(page.locator(path==='/admin'?'.admin-access-portal':'.nopal-header').getByRole('button',{name:lang==='es'?'Español':'English',exact:true})).toHaveAttribute('aria-pressed','true');else await expect(page.getByRole('combobox',{name:'Language / Idioma',exact:true})).toHaveValue(lang);await expect(page.locator('html')).toHaveAttribute('lang',lang==='es'?'es-MX':'en');expect(await page.locator('body').innerText()).not.toMatch(/[→←↑↓↗↘]/);}
 await page.reload();await expect(page.getByRole('combobox',{name:'Language / Idioma',exact:true})).toHaveValue(lang);
 await page.setViewportSize({width:375,height:812});for(const path of ['/','/login','/inquiries','/admin','/explore']){await page.goto(path);if(path==='/'||path==='/admin')await expect(page.locator(path==='/admin'?'.admin-access-portal':'.nopal-header').getByRole('group',{name:'Language / Idioma',exact:true})).toBeVisible();else await expect(page.getByRole('combobox',{name:'Language / Idioma',exact:true})).toBeVisible();expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth)).toBe(true);}
});
test('switching language preserves search, user data and current route; invalid stored language recovers',async({page})=>{
 await page.addInitScript(key=>{if(!localStorage.getItem(key))localStorage.setItem(key,'unsupported');},key);
 await page.route('**/api/public/produce**',r=>r.fulfill({json:{items:[{id:'test',name:'Products',company:'Home',description:'Company',category:'Fruit',country:'Mexico',region:'Jalisco',bio:'Profile',quantity:1.25,unit:'kg',price:12.5,currency:'USD',availability:'Public',business_email:'',business_phone:''}],page:1,hasMore:false}}));
 await page.goto('/explore?q=Products');await expect(page.getByRole('combobox',{name:'Language / Idioma',exact:true})).toHaveValue('en');await page.getByRole('combobox',{name:'Language / Idioma',exact:true}).selectOption('es');
 await expect(page.getByRole('heading',{name:'Explorar productos',exact:true})).toBeVisible();await expect(page.locator('.produce-card h3')).toHaveText('Products');await expect(page.locator('.produce-card strong')).toHaveText('Home');expect(new URL(page.url()).searchParams.get('q')).toBe('Products');await expect(page.getByLabel('Producto o productor')).toHaveValue('Products');
 await page.goto('/login');await expect(page.getByRole('heading',{name:'Bienvenido de nuevo.'})).toBeVisible();await page.reload();await expect(page.getByRole('combobox',{name:'Language / Idioma',exact:true})).toHaveValue('es');
});
