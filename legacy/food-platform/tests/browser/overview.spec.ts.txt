 const urls:string[]=[];await page.route('**/api/admin/**',r=>{urls.push(r.request().url());return r.fulfill({json:new URL(r.request().url()).pathname.endsWith('/operations-overview')?{...brand,recent:snapshot.recent}:new URL(r.request().url()).pathname.endsWith('/overview')?snapshot:{items:[],page:1,hasMore:false}});});await page.goto('/admin');await page.getByRole('combobox',{name:'Language / Idioma',exact:true}).selectOption(language);await expect(page.locator('.overview-kpi')).toHaveCount(5);await expect(page.locator('.overview-kpi strong')).toHaveText(['4','2','6','1','5']);await expect(page.locator('.overview-charts figure')).toHaveCount(2);await expect(page.locator('.overview-bar-group')).toHaveCount(3);await expect(page.locator('.overview-trend li')).toHaveCount(7);await expect(page.locator('.overview-activities')).not.toContainText('private-identifier');await expect(page.locator('.overview-asof')).toContainText('UTC');

test('empty, unqueried, loading and failed Overview are distinct; refresh replaces the snapshot',async({page})=>{
 let mode='loading';await page.route('**/api/admin/operations-overview*',r=>r.fulfill({json:mode==='missing'?{}:mode==='new'?{...brand,recent:snapshot.recent}:{counts:Object.fromEntries(Object.keys(brand.counts).map(k=>[k,0])),recent:[]}}));let release:()=>void=()=>{};await page.route('**/api/admin/overview*',async r=>{if(mode==='loading')await new Promise<void>(resolve=>{release=resolve;});if(mode==='error')return r.fulfill({status:503,json:{error:'Overview unavailable'}});const zero={...snapshot,counts:Object.fromEntries(Object.keys(snapshot.counts).map(k=>[k,0])),approvals:snapshot.approvals.map(r=>({...r,pending:0,approved:0,rejected:0})),inquiries:{received:0,reviewing:0,resolved:0},activityTrend:null,recent:[]};return r.fulfill({json:mode==='missing'?{counts:{products:0},recent:[],health:{}}:mode==='new'?snapshot:zero});});await page.goto('/admin',{waitUntil:'domcontentloaded'});await expect(page.locator('.overview-kpis')).toHaveCount(0);await expect(page.getByRole('status')).toContainText('Checking administrator access and loading records');mode='zero';release();await expect(page.locator('.overview-kpi strong')).toHaveText(['0','0','0','0','0']);await expect(page.getByText('No registrations recorded.',{exact:true})).toBeVisible();await expect(page.getByText('No saved inquiries yet.',{exact:true})).toBeVisible();await expect(page.locator('.overview-trend')).toHaveCount(0);
 mode='missing';await page.getByRole('button',{name:'Refresh records',exact:true}).click();await expect(page.locator('.overview-kpi strong').first()).toHaveText('Not queried');await expect(page.getByText('Approval breakdown was not queried.',{exact:true})).toBeVisible();mode='new';await page.getByRole('button',{name:'Refresh records',exact:true}).click();await expect(page.locator('.overview-kpi strong').first()).toHaveText('4');await expect(page.locator('.overview-trend li')).toHaveCount(7);mode='error';await page.getByRole('button',{name:'Refresh records',exact:true}).click();await expect(page.getByRole('alert')).toContainText('Overview unavailable');await expect(page.locator('.overview-kpis')).toHaveCount(0);
});
