import test from 'node:test';
import assert from 'node:assert/strict';
import {checkStagingReadonly} from '../scripts/check-staging-readonly.mjs';
import {STAGING} from '../scripts/lib/admin-grant.mjs';
const {Response}=globalThis;
const secret='synthetic-sensitive-value';
const bindings=[{name:'DB',type:'d1',id:STAGING.database_id},{name:'AUTH_ENVIRONMENT',text:'staging'},{name:'AUTH_ORIGIN',text:'https://jalisco-auth-staging.yukikato123123123.workers.dev'},{name:'SUPABASE_URL',text:`https://${STAGING.project_ref}.supabase.co`},{name:'SECRET',text:secret}];
test('existing connection diagnostics issue fixed SELECT only and never turn a review file into approval',async()=>{
 const calls=[];
 const r=await checkStagingReadonly({cloudflareToken:secret,approvalFilePresent:true,fetcher:async(url,o)=>{calls.push({url,o});assert.equal(o.redirect,'error');if(o.method==='GET')return Response.json({success:true,result:{bindings}});const b=JSON.parse(o.body);assert.ok(b.sql.startsWith('SELECT '));assert.ok(!/INSERT|UPDATE|DELETE|PRAGMA/i.test(b.sql));return Response.json({success:true,result:[{success:true,results:[{known_user:1,onboarded:1,grant_rows:0,grant_events:0,stored_unexpired_normal_sessions:0,secret}],meta:{rows_written:0,changed_db:false}}]});}});
 assert.equal(calls.length,2);assert.equal(r.application.status,'VERIFIED_READ_ONLY');assert.equal(r.approval_authorized,false);assert.equal(r.supabase.status,'NOT_CONNECTED');assert.equal(r.migration,'NOT_RECHECKED');assert.equal(JSON.stringify(r).includes(secret),false);
});
test('scope mismatch or failed authentication prevents D1 requests and secret-bearing errors stay suppressed',async()=>{
 for(const failed of [true,false]){let n=0;const r=await checkStagingReadonly({cloudflareToken:secret,fetcher:async()=>{n++;if(failed)throw Error(secret);return Response.json({success:true,result:{bindings:[]}});}});assert.equal(n,1);assert.equal(r.application,undefined);assert.equal(JSON.stringify(r).includes(secret),false);}
});
test('Supabase uses official read-only endpoint and identity matching never proves ownership',async()=>{
 const r=await checkStagingReadonly({supabaseToken:secret,fetcher:async(url,o)=>{assert.equal(url,`https://api.supabase.com/v1/projects/${STAGING.project_ref}/database/query/read-only`);assert.equal(o.method,'POST');assert.ok(JSON.parse(o.body).query.startsWith('SELECT '));return Response.json([{id:'11111111-1111-4111-8111-111111111111',email:'administrator@example.test',email_confirmed_at:'2026-01-01',banned_until:null,deleted_at:null}]);}});
 assert.equal(r.supabase.status,'IDENTITY_MATCHED');assert.equal(r.supabase.ownership_independently_verified,false);assert.equal(r.approval_authorized,false);assert.equal(JSON.stringify(r).includes(secret),false);
});
